    // Lọc sản phẩm theo danh mục
    function filterByCategory(categoryName) {
      const buttons = document.querySelectorAll('.cat-filter-btn');
      buttons.forEach(btn => {
        if (btn.getAttribute('data-cat') === categoryName) {
          btn.classList.add('bg-[#c8102e]', 'text-white', 'border-[#c8102e]');
          btn.classList.remove('bg-slate-50', 'text-slate-700', 'border-slate-200');
        } else {
          btn.classList.remove('bg-[#c8102e]', 'text-white', 'border-[#c8102e]');
          btn.classList.add('bg-slate-50', 'text-slate-700', 'border-slate-200');
        }
      });

      if (categoryName === 'all') {
        renderAllGrids(productsData, 'CHÀO MỪNG BẠN ĐẾN VIỄN TÍN AUTO - TẤT CẢ SẢN PHẨM');
      } else {
        const matched = productsData.filter(p => p.category.toLowerCase() === categoryName.toLowerCase());
        const listToShow = matched.length > 0 ? matched : productsData;
        const heading = matched.length > 0
          ? `DANH MỤC: ${categoryName.toUpperCase()}`
          : `DANH MỤC: ${categoryName.toUpperCase()} (GỢI Ý SẢN PHẨM NỔI BẬT)`;
        renderAllGrids(listToShow, heading);
      }
    }

    // Tìm kiếm sản phẩm trên thanh Header
    function handleSearch(e) {
      e.preventDefault();
      const keyword = document.getElementById('searchInput').value.trim().toLowerCase();
      const selectedCat = document.getElementById('searchCategorySelect').value;

      let filtered = productsData;
      if (selectedCat !== 'all') {
        filtered = filtered.filter(p => p.category.toLowerCase() === selectedCat.toLowerCase());
      }
      if (keyword) {
        filtered = filtered.filter(p =>
          p.name.toLowerCase().includes(keyword) ||
          p.category.toLowerCase().includes(keyword)
        );
      }
      renderAllGrids(filtered, keyword ? `KẾT QUẢ TÌM KIẾM: "${keyword.toUpperCase()}"` : 'SẢN PHẨM NỔI BẬT');
      scrollToSection('san-pham-noi-bat');
    }

    // Mở Modal Xem Nhanh
    function openQuickView(productId) {
      const product = productsData.find(p => p.id === productId);
      if (!product) return;
      selectedModalProduct = product;

      document.getElementById('modalProductTitle').textContent = product.name;
      document.getElementById('modalProductFullTitle').textContent = product.name;
      document.getElementById('modalProductCat').textContent = product.category;
      document.getElementById('modalProductOldPrice').textContent = formatVND(product.oldPrice);
      document.getElementById('modalProductSalePrice').textContent = `${formatVND(product.salePrice)} (-${product.discountPercent}%)`;

      const imgEl = document.getElementById('modalProductImg');
      imgEl.src = `https://lh3.googleusercontent.com/d/${product.driveId}`;
      imgEl.onerror = function() {
        this.onerror = null;
        this.src = `https://drive.google.com/thumbnail?id=${product.driveId}&sz=w800`;
      };

      const modal = document.getElementById('quickViewModal');
      modal.classList.remove('hidden');
      modal.classList.add('flex');
    }

    function closeQuickView() {
      const modal = document.getElementById('quickViewModal');
      modal.classList.add('hidden');
      modal.classList.remove('flex');
    }

    function selectProductForQuote() {
      closeQuickView();
      if (selectedModalProduct) {
        document.getElementById('quoteNote').value = `Tôi cần tư vấn & báo giá lắp đặt: ${selectedModalProduct.name} (${formatVND(selectedModalProduct.salePrice)})`;
      }
      scrollToSection('yeu-cau-bao-gia');
      document.getElementById('quoteName').focus();
    }

    // Thêm vào giỏ hàng & chuyển xuống form báo giá nhanh
    function addToCart(productId) {
      cartCount++;
      document.getElementById('cartCountBadge').textContent = cartCount;
      const product = productsData.find(p => p.id === productId);
      if (product) {
        selectedModalProduct = product;
        openQuickView(productId);
      }
    }

    // Điều khiển 2 Banner Tự Trượt (Auto Slider)
    let currentSlideIndex = 0;
    const slides = document.querySelectorAll('.banner-slide');
    const dot0 = document.getElementById('sliderDot0');
    const dot1 = document.getElementById('sliderDot1');

    function showSlide(index) {
      currentSlideIndex = (index + slides.length) % slides.length;
      slides.forEach((slide, i) => {
        if (i === currentSlideIndex) {
          slide.classList.remove('opacity-0', 'z-0');
          slide.classList.add('opacity-100', 'z-10');
        } else {
          slide.classList.remove('opacity-100', 'z-10');
          slide.classList.add('opacity-0', 'z-0');
        }
      });

      if (currentSlideIndex === 0) {
        dot0.className = 'w-5 h-1.5 rounded-full bg-amber-400 transition-all';
        dot1.className = 'w-2 h-1.5 rounded-full bg-white/50 transition-all';
      } else {
        dot0.className = 'w-2 h-1.5 rounded-full bg-white/50 transition-all';
        dot1.className = 'w-5 h-1.5 rounded-full bg-amber-400 transition-all';
      }
    }

    function changeSlide(step) {
      showSlide(currentSlideIndex + step);
    }

    function goToSlide(index) {
      showSlide(index);
    }
